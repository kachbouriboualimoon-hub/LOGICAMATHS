% ==================== EXERCICE 17 ====================
\section*{Exercice 17}

Dans la figure ci-contre, $OABCDEFG$ et $OABCD'E'F'G'$ sont deux cubes identiques d'arête 1. On munit l'espace du repère orthonormé direct $(O, \overrightarrow{OA}, \overrightarrow{OC}, \overrightarrow{OD})$. Les points $K$ et $L$ sont définis par $\overrightarrow{OK} = a\,\overrightarrow{OD}$ et $\overrightarrow{EL} = (1-a)\overrightarrow{OC}$ où $a$ est un réel de l'intervalle $]0,1[$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Donner les coordonnées des points $C$, $B$, $F$ et $K$.
        \item Montrer que $\overrightarrow{BC} \wedge \overrightarrow{BK} = a\,\overrightarrow{OC} + \overrightarrow{OD}$.
        \item Calculer le volume du tétraèdre $FBCK$.
    \end{enumerate}

    \item Soit $P$ le plan $(BCK)$. Montrer qu'une équation de $P$ est $a y + z - a = 0$.

    \item
    \begin{enumerate}
        \item Donner les coordonnées de $E'$. En déduire que $L(1, 1-a, -1)$.
        \item Montrer que $B$ est le projeté orthogonal du point $L$ sur le plan $P$.
    \end{enumerate}

    \item Soit $(S)$ l'ensemble des points $M(x,y,z)$ de l'espace tels que
    \[
        x^2 + y^2 + z^2 - 2x + 2(a-1)y + 2z + 1 - 2a = 0.
    \]
    \begin{enumerate}
        \item Montrer que $(S)$ est la sphère de centre $L$ et de rayon $R = \sqrt{2 + a^2}$.
        \item Montrer que $(S)$ et $P$ se coupent suivant un cercle dont on précisera le centre et le rayon.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 39 ====================
\section*{Exercice 39}

L'espace est rapporté à un repère orthonormé direct $(O, \vec{i}, \vec{j}, \vec{k})$.

On considère les points $A(1,1,1)$, $B(0,4,0)$, $C(0,0,2)$ et $I(-1,1,-1)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer les composantes du vecteur $\overrightarrow{AB} \wedge \overrightarrow{AC}$.
        \item Calculer le volume $V$ du tétraèdre $ABCI$.
    \end{enumerate}

    \item On désigne par $P$ le plan $(ABC)$. Montrer qu'une équation cartésienne de $P$ est $x + y + 2z - 4 = 0$.

    \item Soit $(S)$ l'ensemble des points $M(x,y,z)$ de l'espace tel que
    \[
        x^2 + y^2 + z^2 + 2x - 2y + 2z - 8 = 0.
    \]
    \begin{enumerate}
        \item Montrer que $(S)$ est la sphère de centre $I$ et de rayon $\sqrt{11}$.
        \item Montrer que $P \cap (S)$ est un cercle $(\mathcal{C})$ de rayon $\sqrt{5}$.
        \item Vérifier que le segment $[BC]$ est un diamètre du cercle $(\mathcal{C})$. En déduire les coordonnées du point $H$, centre de $(\mathcal{C})$.
    \end{enumerate}

    \item Soit $a$ un réel et $M$ le point défini par $\overrightarrow{AM} = a\,\overrightarrow{AB}$.
    \begin{enumerate}
        \item Déterminer à l'aide du réel $a$, les coordonnées du point $M$.
        \item Montrer que $\overrightarrow{BM} \cdot \overrightarrow{CM} = (a-1)(11a+3)$.
        \item En déduire que la droite $(AB)$ recoupe le cercle $(\mathcal{C})$ au point $E$ défini par $\overrightarrow{AE} = -\dfrac{3}{11}\overrightarrow{AB}$.
        \item Montrer que le volume $V'$ du tétraèdre $AECI$ est égal à $\dfrac{3}{11} V$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 44 ====================
\section*{Exercice 44}

Dans la figure ci-contre, $ABCDEFGH$ est un cube d'arête 1.
\[
\overrightarrow{AI} = \overrightarrow{EJ} = \frac{\sqrt{2}}{2}\overrightarrow{AD}
\]
On note $\vec{i} = \overrightarrow{AB}$, $\vec{j} = \overrightarrow{AD}$ et $\vec{k} = \overrightarrow{AE}$ et on munit l'espace du repère orthonormé direct $(A, \vec{i}, \vec{j}, \vec{k})$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer les coordonnées des points $F$, $G$, $I$ et $J$.
        \item Justifier qu'une représentation paramétrique de la droite $(IJ)$ est
        \[
            \begin{cases}
                x = 0 \\
                y = \dfrac{\sqrt{2}}{2} \\
                z = \alpha
            \end{cases}
            \quad (\alpha \in \mathbb{R}).
        \]
    \end{enumerate}
    Dans la suite de l'exercice $\alpha$ est un réel et $M$ est un point de la droite $(IJ)$ de coordonnées $\left(0, \dfrac{\sqrt{2}}{2}, \alpha\right)$.

    \item
    \begin{enumerate}
        \item Vérifier que
        \[
            \overrightarrow{AF} \wedge \overrightarrow{AM} = -\frac{\sqrt{2}}{2}\vec{i} - \alpha\vec{j} + \frac{\sqrt{2}}{2}\vec{k}
        \]
        et que
        \[
            \overrightarrow{BC} \wedge \overrightarrow{BM} = \alpha\vec{i} + \vec{k}.
        \]
        \item En déduire que les triangles $AFM$ et $BCM$ ont la même aire.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Montrer que
        \[
            (\overrightarrow{AF} \wedge \overrightarrow{AM}) \cdot \overrightarrow{AG} = -\alpha
        \]
        et que
        \[
            (\overrightarrow{BC} \wedge \overrightarrow{BM}) \cdot \overrightarrow{BG} = 1.
        \]
        \item Montrer que $(M,A,F,G$ sont coplanaires$)$ si et seulement si $(M$ et $I$ sont confondus$)$.
        \item Déterminer les points $M$ de la droite $(IJ)$ pour lesquels $AFMG$ et $BCMG$ sont deux tétraèdres de même volume.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 45 ====================
\section*{Exercice 45}

L'espace est rapporté à un repère orthonormé direct $(O, \vec{i}, \vec{j}, \vec{k})$. On considère les points $A(2,0,0)$, $B(0,4,0)$ et $C(0,0,4)$ et on désigne par $I$ et $J$ les milieux respectifs des segments $[AB]$ et $[BC]$.

\begin{enumerate}
    \item Déterminer les coordonnées des points $I$ et $J$.

    \item Soit $P$ l'ensemble des points $M$ de l'espace vérifiant $MI = MJ$.
    \begin{enumerate}
        \item Montrer que $P$ est le plan d'équation $2x - 4z + 3 = 0$.
        \item Montrer que la droite $(OC)$ et le plan $P$ sont sécants en un point $K$ que l'on précisera.
    \end{enumerate}

    \item Soit $S$ l'ensemble des points $M(x,y,z)$ de l'espace tels que
    \[
        x^2 + y^2 + z^2 - \frac{3}{2}z - 5 = 0.
    \]
    \begin{enumerate}
        \item Montrer que $S$ est une sphère dont on déterminera le centre et le rayon.
        \item Vérifier que les points $I$ et $J$ appartiennent à la sphère $S$.
        \item Montrer que $S$ est la seule sphère qui passe par les points $I$ et $J$ et dont le centre est un point de la droite $(OC)$.
    \end{enumerate}

    \item Déterminer l'intersection du plan $P$ avec la sphère $S$.
\end{enumerate}

% ==================== EXERCICE 49 ====================
\section*{Exercice 49}

L'espace est rapporté à un repère orthonormé direct $(O, \vec{i}, \vec{j}, \vec{k})$. On considère les points $O(0,0,0)$, $B(2,2,-1)$, $C(-2,0,-1)$ et $\Omega(1,-1,-3)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Calculer les composantes du vecteur $\overrightarrow{OB} \wedge \overrightarrow{OC}$.
        \item Soit $P$ le plan contenant les points $O$, $B$ et $C$. Justifier qu'une équation cartésienne du plan $P$ est $x - 2y - 2z = 0$.
        \item Vérifier que le point $\Omega$ n'appartient pas au plan $P$. Calculer le volume du tétraèdre $OBC\Omega$.
    \end{enumerate}

    \item Soit $S$ l'ensemble des points $M(x,y,z)$ de l'espace tels que
    \[
        x^2 + y^2 + z^2 - 2x + 2y + 6z - 3 = 0.
    \]
    \begin{enumerate}
        \item Montrer que $S$ est la sphère de centre $\Omega$ et de rayon $\sqrt{14}$.
        \item Montrer que $S$ et $P$ sont sécants suivant le cercle $\zeta$ de centre $H(0,1,-1)$ et de rayon $\sqrt{5}$.
        \item Montrer que $[BC]$ est un diamètre du cercle $\zeta$.
    \end{enumerate}

    \item Soit $a$ un réel et $P_a$ le plan d'équation
    \[
        ax - 2ay - 2z + 2(a-1) = 0.
    \]
    \begin{enumerate}
        \item Vérifier que pour tout réel $a$, la droite $(BC)$ est contenue dans le plan $P_a$.
        \item Déduire que la sphère $S$ et le plan $P_a$ sont sécants.
        \item On désigne par $r_a$ le rayon du cercle intersection de la sphère $S$ et du plan $P_a$. Sans expliciter $r_a$, justifier que $r_a \geq \sqrt{5}$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 51 ====================
\section*{Exercice 51}

L'espace est rapporté à un repère orthonormé direct $(O, \vec{i}, \vec{j}, \vec{k})$. On considère les points $A(1,1,0)$, $B(0,1,1)$ et $C(-1,5,-2)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer les composantes du vecteur $\overrightarrow{AB} \wedge \overrightarrow{AC}$.
        \item En déduire que les points $A$, $B$ et $C$ déterminent un plan $P$ dont une équation cartésienne est $x + y + z - 2 = 0$.
    \end{enumerate}

    \item Soit $Q$ le plan d'équation $x + y + z + 1 = 0$.
    \begin{enumerate}
        \item Prouver que $P$ et $Q$ sont strictement parallèles.
        \item Soit $I(x,y,z)$ un point du plan $Q$. Montrer que le volume du tétraèdre $IABC$ est égal à $2$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Montrer que le triangle $ABC$ est rectangle en $A$.
        \item Soit $H$ le milieu du segment $[BC]$ et $\Delta$ la droite dont une représentation paramétrique est
        \[
            \begin{cases}
                x = \alpha - \dfrac{1}{2} \\
                y = \alpha + 3 \\
                z = \alpha - \dfrac{1}{2}
            \end{cases}
            \quad (\alpha \in \mathbb{R}).
        \]
        Vérifier que $H$ est un point de $\Delta$ et montrer que $\Delta$ est perpendiculaire à $P$ en $H$.
    \end{enumerate}

    \item Soit $\Omega\left(\alpha - \dfrac{1}{2}, \alpha + 3, \alpha - \dfrac{1}{2}\right)$ un point de $\Delta$. $S_\alpha$ la sphère de centre $\Omega$ et passant par $B$.
    \begin{enumerate}
        \item Montrer que le rayon de la sphère $S_\alpha$ est $R = \sqrt{3\alpha^2 + \dfrac{13}{2}}$.
        \item Montrer que pour tout réel $\alpha$, la sphère $S_\alpha$ passe par le point $C$.
        \item Montrer que pour tout réel $\alpha$, le plan $P$ coupe la sphère $S_\alpha$ selon un cercle $\Gamma$ dont on déterminera le centre et le rayon.
        \item Déterminer le réel $\alpha$ pour que la sphère $S_\alpha$ soit tangente au plan $Q$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 53 ====================
\section*{Exercice 53}

Dans l'espace $(\xi)$ rapporté à un repère orthonormé $(O, \vec{i}, \vec{j}, \vec{k})$, on donne les points $A(-1,-1,1)$, $B(3,2,-1)$ et $C(1,\frac{1}{2},1)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Montrer que les points $A$, $B$ et $C$ ne sont pas alignés.
        \item Vérifier qu'une équation cartésienne du plan $(P)$ défini par les points $A$, $B$ et $C$ est
        \[
            3x - 4y - 1 = 0.
        \]
    \end{enumerate}

    \item Soit $m$ un réel et $(S_m)$ l'ensemble des points $M(x,y,z)$ de l'espace $(\xi)$ tels que
    \[
        x^2 + y^2 + z^2 - m x - 2(m+1)y + m^2 + 2m = 0.
    \]
    \begin{enumerate}
        \item Montrer que $(S_m)$ est une sphère dont on précisera, en fonction de $m$, le rayon $R_m$ et les coordonnées du centre $I_m$.
        \item Déterminer l'ensemble des points $I_m$ lorsque $m$ décrit l'ensemble des réels.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Étudier suivant les valeurs de $m$, la position relative de la sphère $(S_m)$ et du plan $(P)$.
        \item Montrer que l'intersection de $(S_5)$ et $(P)$ est un cercle dont on précisera le centre et le rayon.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 54 ====================
\section*{Exercice 54}

Dans l'espace $(\xi)$ , on considère trois points non alignés $O$ , $A$ , $B$ et $G$ le point défini par :
\[
        \overrightarrow{GO} + 2\overrightarrow{GA} + 3\overrightarrow{GB} = \overrightarrow{0}
    \]
\begin{enumerate}
    \item Vérifier que
    \[
        \overrightarrow{OG} = \frac{1}{3}\overrightarrow{OA} + \frac{1}{2}\overrightarrow{OB}.
    \]

    \item Soit $C$ un point de l'espace $(\xi)$ n'appartenant pas au plan $(OAB)$ et soit $(S)$ l'ensemble des points $M$ de l'espace $(\xi)$ tels que
    \[
        \left(\overrightarrow{MO} + 2\overrightarrow{MA} + 3\overrightarrow{MB}\right) \cdot \overrightarrow{MC} = 0.
    \]
    \begin{enumerate}
        \item Montrer que
        \[
            (M \in (S)) \Longleftrightarrow \left(\overrightarrow{MG} \cdot \overrightarrow{MC} = 0\right).
        \]
        \item En déduire la nature de l'ensemble $(S)$.
    \end{enumerate}

    \item Dans toute la suite, on munit l'espace $(\xi)$ d'un repère orthonormé $(O, \vec{i}, \vec{j}, \vec{k})$. On suppose que les points $A$, $B$ et $C$ ont pour coordonnées respectives $(6,0,0)$, $(0,6,0)$ et $(0,0,4)$.
    \begin{enumerate}
        \item Vérifier que les points $A$, $B$ et $O$ ne sont pas alignés.
        \item Déterminer les coordonnées du point $G$ vérifiant
        \[
            \overrightarrow{GO} + 2\overrightarrow{GA} + 3\overrightarrow{GB} = \overrightarrow{0}.
        \]
        \item Vérifier qu'une équation cartésienne de la sphère $(S)$ est
        \[
            x^2 + y^2 + z^2 - 2x - 3y - 4z = 0.
        \]
        \item Soit $(P)$ le plan dont une équation cartésienne est $z = 0$. Montrer que le plan $(P)$ coupe la sphère $(S)$ suivant le cercle de diamètre $[OG]$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 60 ====================
\section*{Exercice 60}

Dans l'espace muni d'un repère orthonormé $(O, \vec{i}, \vec{j}, \vec{k})$, on considère l'ensemble $(S)$ des points $M(x,y,z)$ tels que
\[
    x^2 + y^2 + z^2 - 4y - 5 = 0.
\]

\begin{enumerate}
    \item Montrer que l'ensemble $(S)$ est une sphère de centre $\Omega(0,2,0)$ et de rayon $3$.

    \item Soit $(P)$ le plan dont une équation cartésienne est
    \[
        2x - 2y + z - 2 = 0.
    \]
    Déterminer la position relative de la sphère $(S)$ et du plan $(P)$. Caractériser $(S) \cap (P)$.

    \item Soit $(P_m)$ le plan dont une équation cartésienne est
    \[
        2m x + (1-2m)y + m z + 1 - 2m = 0.
    \]
    \begin{enumerate}
        \item Soit $(\Delta)$ la droite dont une représentation paramétrique est
        \[
            \begin{cases}
                x = \lambda \\
                y = -1 \\
                z = -2\lambda
            \end{cases}
            \quad (\lambda \in \mathbb{R}).
        \]
        Vérifier que la droite $(\Delta)$ est incluse dans le plan $(P_m)$.
        \item Calculer la distance du point $\Omega$ au plan $(P_m)$.
        \item Déterminer $m$ pour que le plan $(P_m)$ soit tangent à la sphère $(S)$. Préciser les coordonnées du point de contact.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 61 ====================
\section*{Exercice 61}

Dans l'espace rapporté à un repère orthonormé $(O, \vec{i}, \vec{j}, \vec{k})$, on considère les points $A(-1,-1,1)$, $B(-1,2,-2)$ et le plan $(P)$ dont une équation cartésienne est $x + y + z - 2 = 0$.

\begin{enumerate}
    \item Montrer que la droite $(AB)$ est parallèle au plan $(P)$.

    \item Soit $\alpha$ un réel et $(S_\alpha)$ l'ensemble des points $M(x,y,z)$ de l'espace tels que
    \[
        x^2 + y^2 + z^2 + 2x - 2\alpha y + 2\alpha z + \alpha^2 + \alpha = 0.
    \]
    \begin{enumerate}
        \item Montrer que pour tout réel $\alpha$, l'ensemble $(S_\alpha)$ est une sphère de centre $I_\alpha(-1,\alpha,-\alpha)$ et dont on précisera le rayon.
        \item Montrer que, quand $\alpha$ décrit l'ensemble des réels, le point $I_\alpha$ décrit la droite $(AB)$.
    \end{enumerate}

    \item Étudier suivant les valeurs du réel $\alpha$ les positions relatives de la sphère $(S_\alpha)$ et du plan $(P)$.

    \item Soit $I$ le milieu du segment $[AB]$ et $I_{1-\alpha}$ le centre de la sphère $(S_{1-\alpha})$.
    \begin{enumerate}
        \item Montrer que le point $I$ est le milieu du segment $[I_\alpha,\, I_{1-\alpha}]$.
        \item En déduire que les sphères $(S_\alpha)$ et $(S_{1-\alpha})$ sont symétriques par rapport au point $I$.
    \end{enumerate}
\end{enumerate}
