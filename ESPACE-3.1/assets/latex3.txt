% ==================== EXERCICE 5 ====================
\section*{Exercice 5}

L'espace est muni d'un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On donne les points $A(1,0,1)$, $B(0,-2,1)$ et $C(0,0,2)$ et le vecteur $\vec{N}\begin{pmatrix}2\\-1\\2\end{pmatrix}$..

\begin{enumerate}
    \item Vérifier que les points $A$, $B$ et $C$ déterminent un plan $P$.

    \item
    \begin{enumerate}
        \item Vérifier que le vecteur $\overrightarrow{N}$ est orthogonal à chacun des vecteurs $\overrightarrow{AB}$ et $\overrightarrow{AC}$.
        \item En déduire qu'une équation cartésienne du plan $P$ est : $2x - y + 2z - 4 = 0$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Montrer que le point $E(0,0,1)$ n'appartient pas au plan $P$.
        \item Montrer que le point $H\left(\frac{4}{9}, -\frac{2}{9}, \frac{13}{9}\right)$ est le projeté orthogonal du point $E$ sur $P$.
        \item Montrer que le point $H$ est l'orthocentre du triangle $ABC$.
    \end{enumerate}

    \item Soit $S$ l'ensemble des points $M(x,y,z)$ du plan tel que : $x^2 + y^2 + z^2 - 2z = 0$.
    \begin{enumerate}
        \item Montrer que $S$ est la sphère de centre $E$ et de rayon $1$.
        \item Montrer que $S$ et $P$ se coupent suivant un cercle $(\zeta)$ dont on précisera le centre et le rayon.
        \item Montrer que $(\zeta)$ passe par les points $A$ et $C$.
        \item Déduire que le plan $(EHB)$ est le plan médiateur du segment $[AC]$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 6 ====================
\section*{Exercice 6}

L'espace est muni d'un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On considère les points $A(2,0,2)$, $B(2,1,1)$ , $C(1,2,1)$ et $E(-1,-1,0)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Montrer que $\overrightarrow{AB} \wedge \overrightarrow{AC} = \vec{i} + \vec{j} + \vec{k}$.
        \item Montrer que l'aire du triangle $ABC$ est égale à $\frac{\sqrt{3}}{2}$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Montrer que les points $A$, $B$, $C$ et $E$ sont non coplanaires.
        \item Calculer le volume du tétraèdre $EABC$.
        \item Montrer que la distance du point $E$ au plan $(ABC)$ est égale à $2\sqrt{3}$.
    \end{enumerate}

    \item Soit $\Delta$ la droite passant par $E$ et perpendiculaire au plan $(ABC)$.
    \begin{enumerate}
        \item Vérifier que le système
        \[
            \begin{cases}
                x = -1 + \alpha \\
                y = -1 + \alpha \\
                z = \alpha
            \end{cases}
            \quad (\alpha \in \mathbb{R})
        \]
        est une représentation paramétrique de $\Delta$.
        \item Vérifier que le point $I(1,1,2)$ appartient à $\Delta$.
        \item Montrer que le point $I$ est le centre du cercle $(\Gamma)$ circonscrit au triangle $ABC$.
    \end{enumerate}

    \item Soit $S$ l'ensemble des points $M(x,y,z)$ de l'espace vérifiant : $x^2 + y^2 + z^2 + 2x + 2y - 12 = 0$.
    \begin{enumerate}
        \item Montrer que $S$ est la sphère de centre $E$ et de rayon $\sqrt{14}$.
        \item Montrer que le plan $(ABC)$ coupe la sphère $S$ suivant le cercle $\Gamma$.
    \end{enumerate}
\item Soient $F$ le point défini par  $\overrightarrow{IF}=\frac{1}{2} \overrightarrow{EF}$ et $S'$ la sphère de centre $F$ et de rayon $\sqrt(14)$. Montrer que le plan $(ABC)$ coupe la sphère $S'$ suivant le cercle $\Gamma$.
\end{enumerate}

% ==================== EXERCICE 10 ====================
\section*{Exercice 10}

L'espace est rapporté à un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On considère les points $A(1,0,2)$, $B(-2,1,-1)$ et $C(0,0,1)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer les composantes du vecteur $\overrightarrow{AB} \wedge \overrightarrow{AC}$.
        \item Déduire que les points $A$, $B$ et $C$ déterminent un plan $P$ dont une équation cartésienne est $x - z + 1 = 0$.
    \end{enumerate}

    \item On considère les points $I(1,-1,-1)$ et $J\left(-\frac{1}{2}, -1, \frac{1}{2}\right)$ et soit $\Delta$ la droite passant par $I$ et perpendiculaire à $P$.
    \begin{enumerate}
        \item Montrer que la droite $\Delta$ coupe le plan $P$ en $J$.
        \item Calculer la distance $IJ$.
    \end{enumerate}

    \item Soit $S$ l'ensemble des points $M(x,y,z)$ de l'espace vérifiant :
    \[
        x^2 + y^2 + z^2 - 2x + 2y + 2z - 2 = 0.
    \]
    \begin{enumerate}
        \item Montrer que $S$ est une sphère de centre $I$ et de rayon $R$ que l'on déterminera.
        \item Montrer que le plan $P$ coupe la sphère $S$ suivant le cercle de centre $J$ et de rayon $r = \frac{1}{\sqrt{2}}$.
    \end{enumerate}

    \item Pour $\theta \in [0, 2\pi[$, on considère le point $N(1 + \cos\theta, -1 + \sin\theta, -3)$.
    \begin{enumerate}
        \item Vérifier que $N$ est un point de la sphère $S$.
        \item Justifier que le point $N$ n'appartient pas au plan $P$.
        \item Montrer que $(\overrightarrow{AB} \wedge \overrightarrow{AC}) \cdot \overrightarrow{AN} = -5 - \cos\theta$.
        \item En déduire la valeur de $\theta$ pour laquelle le volume du tétraèdre $ABCN$ est minimal.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 11 ====================
\section*{Exercice 11}

L'espace est muni d'un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On donne les points $A(2,0,1)$, $B(0,2,1)$ et $C(1,2,0)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer les composantes du vecteur $\overrightarrow{AB} \wedge \overrightarrow{AC}$.
        \item Déduire que les points $A$, $B$ et $C$ déterminent un plan $P$ dont une équation cartésienne est $x + y + z - 3 = 0$.
    \end{enumerate}

    \item Soit la sphère $S$ d'équation $x^2 + y^2 + z^2 = 5$.
    \begin{enumerate}
        \item Vérifier que $A$, $B$ et $C$ sont des points de la sphère $S$.
        \item Déduire alors l'intersection de la sphère $S$ avec le plan $P$.
    \end{enumerate}

    \item Soit le point $D$ de coordonnées $\left(\sqrt{\frac{5}{3}}, \sqrt{\frac{5}{3}}, \sqrt{\frac{5}{3}}\right)$. On désigne par $Q$ le plan passant par $D$ et parallèle au plan $P$.
    \begin{enumerate}
        \item Déterminer une équation cartésienne du plan $Q$.
        \item Montrer que $Q$ est tangent à la sphère $S$ au point $D$.
    \end{enumerate}

    \item Soit $M(x,y,z)$ un point de l'espace n'appartenant pas à $P$.
    \begin{enumerate}
        \item Calculer $(\overrightarrow{AB} \wedge \overrightarrow{AC}) \cdot \overrightarrow{AM}$.
        \item Montrer que le volume $V$ du tétraèdre $MABC$ est égal à $\frac{|x+y+z-3|}{3}$.
        \item En déduire que pour tout point $M$ du plan $Q$, $V = \sqrt{\frac{5}{3}} - 1$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 22 ====================
\section*{Exercice 22}

L'espace est muni d'un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On considère les points $A(1,1,0)$, $B(1,-1,2)$, $C(0,1,1)$ et $D(1,1,4)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Montrer que $A$, $B$ et $C$ déterminent un plan qu'on notera $(P)$.
        \item Justifier que $(P)$ est d'équation $x + y + z - 2 = 0$.
        \item Vérifier que $D$ n'appartient pas au plan $(P)$.
    \end{enumerate}

    \item Soit $\mathcal{E}$ le cercle circonscrit au triangle $ABC$ et $H$ le milieu du segment $[AB]$.
    \begin{enumerate}
        \item Montrer que le triangle $ABC$ est rectangle en $C$.
        \item En déduire que $H$ est le centre du cercle $\mathcal{E}$.
    \end{enumerate}

    \item Soit $\Delta$ la droite perpendiculaire au plan $(P)$ passant par le point $H$. Justifier qu'une représentation paramétrique de $\Delta$ est
    \[
        \begin{cases}
            x = 1 + \alpha \\
            y = \alpha \\
            z = 1 + \alpha
        \end{cases}
        \quad (\alpha \in \mathbb{R}).
    \]

    \item Soit $M$ un point de $\Delta$.
    \begin{enumerate}
        \item Justifier que $MA = MB = MC$.
        \item Montrer qu'il existe un unique point $I$ de $\Delta$ tel que $IA = ID$. Donner ses coordonnées.
        \item Déduire de ce qui précède, que les points $A$, $B$, $C$ et $D$ appartiennent à une même sphère $(S)$ dont on précisera le centre et le rayon.
    \end{enumerate}
\end{enumerate}

L'espace est rapporté à un repère orthonormé $(O,\vec{i},\vec{j},\vec{k})$. On considère la sphère $(S)$ d'équation $x^2 + y^2 + z^2 - 8 = 0$ et le plan $P$ d'équation $x + 2y + z - 6 = 0$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer le centre et le rayon de la sphère $(S)$.
        \item Montrer que le plan $P$ coupe la sphère $(S)$ suivant un cercle $(\mathcal{C})$ dont on précisera le centre et le rayon.
    \end{enumerate}

    \item On donne les points $A(2,0,2)$ et $B(2,2,0)$.
    \begin{enumerate}
        \item Vérifier que $A$ appartient à la sphère $(S)$ et n'appartient pas au plan $P$ et que $B$ appartient au cercle $(\mathcal{C})$.
        \item Soit $Q$ l'ensemble des points $M(x,y,z)$ de l'espace tels que $MA = MB$. Montrer que $Q$ est le plan d'équation $y = z$.
        \item Montrer que les plans $P$ et $Q$ se coupent suivant la droite $\Delta$ dont une représentation paramétrique est
        \[
            \begin{cases}
                x = 6 - 3\alpha \\
                y = \alpha \\
                z = \alpha
            \end{cases}
            \quad (\alpha \in \mathbb{R}).
        \]
    \end{enumerate}

    \item Déterminer un point $C$ du cercle $(\mathcal{C})$ tel que $ABC$ est un triangle équilatéral.
\end{enumerate}

% ==================== EXERCICE 27 ====================
\section*{Exercice 27}

L'espace $\mathcal{E}$ est rapporté à un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On donne les points $A(0,0,1)$, $B(2,1,1)$ et $C(-1,0,2)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Calculer les composantes du vecteur $\overrightarrow{AB} \wedge \overrightarrow{AC}$. En déduire que les points $A$, $B$ et $C$ définissent un plan $P$.
        \item Montrer qu'une équation cartésienne du plan $P$ est $x - 2y + z - 1 = 0$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Justifier que le point $I(1,0,-2)$ n'appartient pas au plan $P$.
        \item Calculer le volume du tétraèdre $IABC$.
    \end{enumerate}

    \item Soit $S$ l'ensemble des points $M(x,y,z)$ de l'espace tels que
    \[
        x^2 + y^2 + z^2 - 2x + 4z + 4 = 0.
    \]
    \begin{enumerate}
        \item Montrer que $S$ est la sphère de centre $I(1,0,-2)$ et de rayon $R = 1$.
        \item Vérifier que le point $H\left(\frac{4}{3}, -\frac{2}{3}, -\frac{5}{3}\right)$ est le projeté orthogonal du point $I$ sur le plan $P$.
        \item Montrer que le plan $P$ coupe la sphère $S$ suivant un cercle $\mathcal{C}$ que l'on caractérisera.
    \end{enumerate}

    \item Soit $S'$ la sphère de centre $I'\left(\frac{1}{2}, 1, -\frac{5}{2}\right)$ et de rayon $R' = \frac{3\sqrt{2}}{2}$.
    \begin{enumerate}
        \item Montrer qu'une équation cartésienne de $S'$ est
        \[
            x^2 + y^2 + z^2 - x - 2y + 5z + 3 = 0.
        \]
        \item Montrer que
        \[
            M(x,y,z) \in S \cap S' \quad \text{si et seulement si} \quad
            \begin{cases}
                x^2 + y^2 + z^2 - 2x + 4z + 4 = 0 \\
                x - 2y + z - 1 = 0
            \end{cases}
        \]
        \item Déduire $S \cap S'$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 28 ====================
\section*{Exercice 28}

L'espace est rapporté à un repère orthonormé direct $(O,\vec{i},\vec{j},\vec{k})$. On donne les points $A(1,2,1)$, $B(1,0,-1)$ et $C(3,2,1)$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer $\overrightarrow{AB} \wedge \overrightarrow{AC}$. En déduire que les points $A$, $B$ et $C$ ne sont pas alignés.
        \item Soit $P$ le plan passant par $A$, $B$ et $C$. Montrer qu'une équation de $P$ est $y - z - 1 = 0$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Calculer $\overrightarrow{AB} \cdot \overrightarrow{AC}$. En déduire que le triangle $ABC$ est rectangle en $A$.
        \item Soit $I$ le milieu du segment $[BC]$. Vérifier que $I(2,1,0)$.
    \end{enumerate}

    \item On considère dans l'espace l'ensemble $S$ des points $M(x,y,z)$ tels que
    \[
        x^2 + y^2 + z^2 - 4x - 2y + 2 = 0.
    \]
    \begin{enumerate}
        \item Montrer que $S$ est la sphère de centre $I$ et de rayon $\sqrt{3}$.
        \item Soit $(\zeta)$ le cercle du plan $P$ de centre $I$ et de rayon $\sqrt{3}$. Montrer que $P$ coupe $S$ suivant $(\zeta)$.
        \item Montrer que le cercle $(\zeta)$ est circonscrit au triangle $ABC$.
    \end{enumerate}

    \item On considère la droite $\Delta : 
    \begin{cases}
        x = 2 \\
        y = -\alpha + 1 \\
        z = \alpha
    \end{cases}
    \quad (\alpha \in \mathbb{R})$.
    \begin{enumerate}
        \item Montrer que $\Delta$ est perpendiculaire à $P$ en $I$.
        \item Soit $\alpha \in \mathbb{R}$. On note $M_\alpha(2,1-\alpha,\alpha)$ un point de $\Delta$. Montrer que $d(M_\alpha, P) = \sqrt{2}|\alpha|$.
        \item Pour $\alpha \neq 0$, on note $V_\alpha$ le volume du tétraèdre $ABCM_\alpha$. Montrer que $V_\alpha = \frac{4}{3}|\alpha|$.
        \item Déterminer les valeurs de $\alpha$ pour lesquelles $V_\alpha = \frac{2\sqrt{6}}{3}$.
        \item Montrer que, pour les valeurs de $\alpha$ trouvées dans la question précédente, $M_\alpha \in S$.
    \end{enumerate}
\end{enumerate}

% ==================== EXERCICE 41 ====================
\section*{Exercice 41}

L'espace est rapporté à un repère orthonormé direct $(O,\vec{u},\vec{v},\vec{w})$. Soit $OADBCEFG$ le cube tel que $\overrightarrow{OA} = \vec{u}$, $\overrightarrow{OB} = \vec{v}$ et $\overrightarrow{OC} = \vec{w}$. On désigne par $I$ et $J$ les milieux respectifs des segments $[AF]$ et $[CG]$.

\begin{enumerate}
    \item
    \begin{enumerate}
        \item Déterminer les coordonnées des points $E$, $I$ et $J$.
        \item Vérifier que $\overrightarrow{OI} \wedge \overrightarrow{OJ} = \frac{1}{4} (\vec{u} - 4\vec{v} + 2\vec{w})$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item Calculer l'aire du triangle $OIJ$.
        \item Calculer le volume du tétraèdre $OIJE$.
        \item La droite passant par $E$ et perpendiculaire au plan $(OIJ)$ coupe le plan $(OIJ)$ en un point $H$. Sans calculer les coordonnées de $H$, justifier que $EH = \frac{\sqrt{21}}{7}$.
    \end{enumerate}

    \item Soit $(S)$ l'ensemble des points $M(x,y,z)$ de l'espace tels que
    \[
        x^2 + y^2 + z^2 - 2x - 2z + \frac{11}{7} = 0.
